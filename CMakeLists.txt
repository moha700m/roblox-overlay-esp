cmake_minimum_required(VERSION 3.15)
project(rbx-esp CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(rbx-esp src/main.cpp)
if(MSVC)
  target_compile_options(rbx-esp PRIVATE /W3 /O2 /EHsc /MT)
  target_link_options(rbx-esp PRIVATE /SUBSYSTEM:CONSOLE)
else()
  find_library(GDI32 gdi32 REQUIRED)
  find_library(USER32 user32 REQUIRED)
endif()
